\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm]
  \def\R{1.6}\def\n{16}\pgfmathsetmacro{\th}{360/\n}
  % left: circle cut in 16 wedges
  \draw[fig] (0,0) circle[radius=\R];
  \foreach \k in {0,...,15} \draw[thin] (0,0) -- (\k*\th:\R);
  \draw[thick] (0,0) -- (0:\R); \node[right] at (\R,0) {$r$};
  \draw[->,thick] (2.45,0) -- (3.15,0);
  % right: wedges rearranged
  \pgfmathsetmacro{\w}{2*\R*sin(\th/2)}
  \pgfmathsetmacro{\hh}{\R*cos(\th/2)}
  \begin{scope}[shift={(4.3,-0.8)}]
    \foreach \k in {0,2,...,14} {
      \pgfmathsetmacro{\xa}{\k*\w/2}
      \fill[gray!30] (\xa,0) -- ++(90-\th/2:\R) arc[start angle=90-\th/2,end angle=90+\th/2,radius=\R] -- cycle;
      \draw[thin] (\xa,0) -- ++(90-\th/2:\R) arc[start angle=90-\th/2,end angle=90+\th/2,radius=\R] -- cycle;}
    \foreach \k in {1,3,...,15} {
      \pgfmathsetmacro{\xa}{\k*\w/2}
      \draw[thin] (\xa,\hh) -- ++(-90-\th/2:\R) arc[start angle=-90-\th/2,end angle=-90+\th/2,radius=\R] -- cycle;}
    \pgfmathsetmacro{\xe}{15*\w/2+\R*sin(\th/2)}
    \draw[decorate,decoration={brace,mirror,amplitude=4pt}] (0,-0.28) -- (\xe,-0.28) node[midway,below=5pt] {about half the circumference: $\pi r$};
    \draw[<->,thick] (\xe+0.35,0) -- (\xe+0.35,\R) node[midway,right] {$r$};
  \end{scope}
\end{tikzpicture}
